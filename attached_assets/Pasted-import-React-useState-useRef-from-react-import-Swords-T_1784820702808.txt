          newBracket.rightQuarts = newBracket.rightQuarts.map(m => m.id === match.id ? { ...m, score1: s1, score2: s2, winner: winnerName } : m);

          newBracket.leftSemis = newBracket.leftSemis.map(m => m.id === match.id ? { ...m, score1: s1, score2: s2, winner: winnerName } : m);
          newBracket.final.p1 = winnerName;
        } else if (roundKey === 'rightSemis') {
          newBracket.rightSemis = newBracket.rightSemis.map(m => m.id === match.id ? { ...m, score1: s1, score2: s2, winner: winnerName } : m);
          newBracket.final.p2 = winnerName;
        } else if (roundKey === 'final') {
          newBracket.final = { ...newBracket.final, score1: s1, score2: s2, winner: winnerName };
        }
        return { ...t, bracket: newBracket };
      }
      return t;

              <button onClick={() => setIsSuperAdminAddGameOpen(false)} className="text-slate-400">✕</button>
            </div>
            <form onSubmit={handleAdminAddGame} className="space-y-3">
              <input 
                type="text" 
                placeholder="Nom du jeu..." 
                value={adminGameForm.name} 
                onChange={(e) => setAdminGameForm({...adminGameForm, name: e.target.value})} 
                required 
                className="w-full bg-slate-950 border border-white/10 rounded-xl px-3 py-2 text-xs text-white" 
              />
              <button type="submit" className="w-full bg-fuchsia-500 text-white font-extrabold py-3 rounded-xl text-xs uppercase cursor-pointer">Publier</button>
            </form>
          </div>
        </div>
      )}

      {isFullHistoryModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/90 backdrop-blur-md animate-in fade-in duration-200">
          <div className="w-full max-w-lg bg-slate-900 border border-white/15 rounded-3xl p-6 space-y-4 shadow-2xl max-h-[85vh] flex flex-col">
            <div className="flex items-center justify-between border-b border-white/10 pb-3">
              <h3 className="font-black text-sm uppercase text-white">Historique Intégral</h3>
              <button onClick={() => setIsFullHistoryModalOpen(false)} className="text-slate-400">✕</button>
            </div>
            <div className="flex-1 overflow-y-auto space-y-2 pr-1">
              {allMatches.map((m) => (
                <div key={m.id} className="p-3 rounded-xl bg-slate-950 border border-white/5 flex items-center justify-between text-xs">
                  <div>
                    <span className="font-bold text-cyan-400">{m.category}</span>
                    <p className="text-white font-semibold mt-0.5">{m.p1} vs {m.p2}</p>
                  </div>
                  <span className={`font-mono font-black ${m.winner === user.name ? 'text-emerald-400' : 'text-rose-400'}`}>
                    {m.winner === user.name ? 'VICTOIRE' : 'DÉFAITE'}
                  </span>
                </div>
              ))}
            </div>
          </div>
        </div>
      )}

      {/* BOTTOM NAVIGATION BAR (3 ET 3 ÉQUILIBRÉS) */}
      <nav className="fixed bottom-0 left-0 right-0 z-40 bg-[#07070A]/95 backdrop-blur-xl border-t border-white/10 px-4 py-2.5">
        <div className="max-w-md mx-auto flex items-center justify-around">
          
          <button 
            onClick={() => setCurrentScreen('home')}
            className={`flex flex-col items-center space-y-1 transition cursor-pointer ${currentScreen === 'home' ? 'text-white' : 'text-slate-500 hover:text-slate-300'}`}
          >
            <Swords className="w-5 h-5" style={{ color: currentScreen === 'home' ? user.themeColor.hex : undefined }} />
            <span className="text-[9px] font-bold uppercase tracking-wider">Accueil</span>
          </button>

          <button 
            onClick={() => setCurrentScreen('leagues')}
            className={`flex flex-col items-center space-y-1 transition cursor-pointer ${currentScreen === 'leagues' ? 'text-white' : 'text-slate-500 hover:text-slate-300'}`}
          >
            <ShieldCheck className="w-5 h-5" style={{ color: currentScreen === 'leagues' ? user.themeColor.hex : undefined }} />
            <span className="text-[9px] font-bold uppercase tracking-wider">Ligue</span>
          </button>

          <button 
            onClick={() => setCurrentScreen('tournaments')}
            className={`flex flex-col items-center space-y-1 transition cursor-pointer ${currentScreen === 'tournaments' ? 'text-white' : 'text-slate-500 hover:text-slate-300'}`}
          >
            <Trophy className="w-5 h-5" style={{ color: currentScreen === 'tournaments' ? user.themeColor.hex : undefined }} />
            <span className="text-[9px] font-bold uppercase tracking-wider">Coupe</span>
          </button>

          <button 
            onClick={() => setCurrentScreen('games')}
            className={`flex flex-col items-center space-y-1 transition cursor-pointer ${currentScreen === 'games' ? 'text-white' : 'text-slate-500 hover:text-slate-300'}`}
          >
            <Gamepad2 className="w-5 h-5" style={{ color: currentScreen === 'games' ? user.themeColor.hex : undefined }} />
            <span className="text-[9px] font-bold uppercase tracking-wider">Jeux</span>
          </button>

          <button 
            onClick={() => setCurrentScreen('stats')}
            className={`flex flex-col items-center space-y-1 transition cursor-pointer ${currentScreen === 'stats' ? 'text-white' : 'text-slate-500 hover:text-slate-300'}`}
          >
            <BarChart3 className="w-5 h-5" style={{ color: currentScreen === 'stats' ? user.themeColor.hex : undefined }} />
            <span className="text-[9px] font-bold uppercase tracking-wider">Stats</span>
          </button>

          <button 
            onClick={() => { setCurrentScreen('profile'); setIsSettingsOpen(false); }}
            className={`flex flex-col items-center space-y-1 transition cursor-pointer ${currentScreen === 'profile' ? 'text-white' : 'text-slate-500 hover:text-slate-300'}`}
          >
            <User className="w-5 h-5" style={{ color: currentScreen === 'profile' ? user.themeColor.hex : undefined }} />
            <span className="text-[9px] font-bold uppercase tracking-wider">Profil</span>
          </button>

        </div>
      </nav>

    </div>
  );
}
